local SaveFileName = "VantablackHub_KeyData.json"

-- Hàm kiểm tra xem key đã lưu còn hạn 24 giờ hay không
local function HasValidSavedKey()
    if readfile and isfile and isfile(SaveFileName) then
        local success, data = pcall(function()
            return game:GetService("HttpService"):JSONDecode(readfile(SaveFileName))
        end)
        if success and data and data.Key == CorrectKey and data.Time then
            -- 86400 giây tương đương với 24 giờ
            if os.time() - data.Time < 86400 then
                return true
            end
        end
    end
    return false
end

-- Hàm lưu key khi nhập đúng
local function SaveKey()
    if writefile then
        local data = {
            Key = CorrectKey,
            Time = os.time()
        }
        pcall(function()
            writefile(SaveFileName, game:GetService("HttpService"):JSONEncode(data))
        end)
    end
end

-- Hiệu ứng lắc bảng khi sai key
local function ShakeWindow()
    local originalPos = UDim2.new(0.5, -210, 0.5, -135)
    local tweenInfo = TweenInfo.new(0.05, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut)
    local ts = game:GetService("TweenService")
    
    ts:Create(MainFrame, tweenInfo, {Position = originalPos + UDim2.new(0, -8, 0, 0)}):Play()
    task.wait(0.06)
    ts:Create(MainFrame, tweenInfo, {Position = originalPos + UDim2.new(0, 8, 0, 0)}):Play()
    task.wait(0.06)
    ts:Create(MainFrame, tweenInfo, {Position = originalPos + UDim2.new(0, -5, 0, 0)}):Play()
    task.wait(0.06)
    ts:Create(MainFrame, tweenInfo, {Position = originalPos + UDim2.new(0, 5, 0, 0)}):Play()
    task.wait(0.06)
    ts:Create(MainFrame, tweenInfo, {Position = originalPos}):Play()
end
